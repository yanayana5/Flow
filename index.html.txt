<script>
(function () {
  // GLOBAL STATE
  const state = {
    income: 0,
    varIncome: 0,
    opex: 0,
    balance: 0,
    adults: 2,
    kids: 1,
    stabilizationDebt: 0,
    shield: 0,
    freeCashFlow: 0,
    goals: []
  };

  // Helper: format RSD nice
  function fmt(v) {
    return Number(v || 0).toLocaleString('sr-RS');
  }

  // CFO INIT
  function initializeCFO() {
    state.income = Number(document.getElementById('input-income').value) || 0;
    state.varIncome = Number(document.getElementById('input-variable-income').value) || 0;
    state.opex = Number(document.getElementById('input-opex').value) || 0;
    state.balance = Number(document.getElementById('input-balance').value) || 0;
    state.adults = Number(document.getElementById('input-adults').value) || 1;
    state.kids = Number(document.getElementById('input-kids').value) || 0;
    state.stabilizationDebt = Number(document.getElementById('input-minus').value) || 0;
    state.shield = Number(document.getElementById('input-shield').value) || 0;

    // Fleksibilna dobit = korporativna profitna margina domaćinstva
    state.freeCashFlow = (state.income + state.varIncome) - state.opex;

    // Inicijalni ciljevi
    state.goals = [];

    if (state.stabilizationDebt > 0) {
      state.goals.push({
        id: genId(),
        name: "🎯 STABILIZACIJA (vraćanje duga/minusa na 0)",
        target: state.stabilizationDebt,
        current: 0,
        months: 6,
        priority: "HITNO"
      });
    }

    // Demo cilj
    state.goals.push({
      id: genId(),
      name: "✈️ Porodično putovanje / rezerva",
      target: 200000,
      current: 40000,
      months: 5,
      priority: "NORMALNO"
    });

    renderDashboard();
    renderGoals();
    calculatePersonalReserve();
    evaluateFinancialHealth();

    document.getElementById('setup-card').classList.add('hidden');
    document.getElementById('main-interface').classList.remove('hidden');
  }

  // Simple id
  function genId() {
    return 'g_' + Math.random().toString(36).slice(2);
  }

  // Dashboard rendering
  function renderDashboard() {
    document.getElementById('lbl-balance').textContent = fmt(state.balance) + " RSD";
    document.getElementById('lbl-shield').textContent = fmt(state.shield) + " RSD";
    document.getElementById('lbl-opex').textContent = fmt(state.opex) + " RSD";
    document.getElementById('lbl-free-cash').textContent = fmt(state.freeCashFlow) + " RSD";
  }

  // Personal reserve
  function calculatePersonalReserve() {
    const factorMin = 3 + state.kids * 0.5;
    const factorComf = 6 + state.kids * 1.0;
    const minReserve = state.opex * factorMin;
    const comfReserve = state.opex * factorComf;

    document.getElementById('lbl-min-reserve').textContent = fmt(minReserve) + " RSD";
    document.getElementById('lbl-comf-reserve').textContent = fmt(comfReserve) + " RSD";

    const exp = document.getElementById('reserve-explanation');
    exp.innerHTML = `
      <strong>Kako smo došli do ovog broja?</strong><br>
      Tvoje domaćinstvo ima <strong>${state.adults}</strong> odrasle osobe i
      <strong>${state.kids}</strong> dece, sa osnovnim OPEX‑om od
      <strong>${fmt(state.opex)} RSD</strong>.<br>
      • <strong>Minimalna rezerva</strong> pokriva oko
      <strong>${factorMin.toFixed(1)} meseci</strong> čistog preživljavanja (bez luksuza).<br>
      • <strong>Komforna rezerva</strong> pokriva oko
      <strong>${factorComf.toFixed(1)} meseci</strong> visoke otpornosti, uključujući troškove škole,
      hrane i nepredviđenog zdravstva za decu.<br><br>
      Cilj nije da se „stidiš“ ako trenutno nemaš ovoliko, već da imaš jasan reper
      gde želiš da stigneš.
    `;
  }

  // Financial health
  function evaluateFinancialHealth() {
    const box = document.getElementById('health-indicator');
    const totalIncome = state.income + state.varIncome;
    const debtToIncome = totalIncome > 0 ? state.stabilizationDebt / totalIncome : 0;
    const reserveMonths = state.opex > 0 ? state.shield / state.opex : 0;

    if (state.stabilizationDebt > 200000 || reserveMonths < 1) {
      box.className = "advisor-box danger";
      box.innerHTML = `
        <div class="health-label">🔴 VISOK PRIORITET STABILIZACIJE</div>
        <div>
          Tvoj zaštitni fond je <strong>kritično nizak</strong> u odnosu na mesečni OPEX, a minus
          stvara dodatni pritisak. Pre nego što otvoriš nove ciljeve „želja“, ima smisla da
          većinu slobodne dobiti preusmeriš u <strong>Stabilizacioni fond</strong> i
          minimalnu rezervu.<br><br>
          Ovo nije kritika, već signal gde je najpametnije da usmeriš energiju u narednih par meseci.
        </div>
      `;
    } else if (reserveMonths >= 1 && reserveMonths < 3) {
      box.className = "advisor-box warning";
      box.innerHTML = `
        <div class="health-label">🟠 POTREBNO POBOLJŠANJE</div>
        <div>
          Sistem je <strong>stabilan</strong>, ali nema dovoljno jastuka za veće šokove.
          Održivost dugoročnih ciljeva je <strong>umerena</strong>: nije drama, ali
          bilo bi zdravo da bar deo slobodne dobiti ide u jačanje Zaštitnog fonda.<br><br>
          Mali pomaci (npr. dodatna gig nedeljno ili smanjenje skrivene potrošnje)
          prave veliku razliku u otpornosti.
        </div>
      `;
    } else {
      box.className = "advisor-box success";
      box.innerHTML = `
        <div class="health-label">🟢 DOBRA OTPORNOST</div>
        <div>
          Čestitamo. Domaćinstvo posluje sa pozitivnom marginom, a postojeća rezerva
          uspešno amortizuje spoljne šokove.<br><br>
          Sledeći pametan potez je da deo te snage usmeriš ka ciljevima
          visokog prioriteta (npr. Stabilizacija, porezi, edukacija dece), a zatim
          postepeno i ka „side quest“ ciljevima (putovanja, renoviranja).
        </div>
      `;
    }
  }

  // Goals rendering
  function priorityBadgeClass(p) {
    if (p === "HITNO") return "badge-hitno";
    if (p === "VAŽNO") return "badge-vazno";
    return "badge-normalno";
  }

  function renderGoals() {
    const container = document.getElementById('goals-container');
    container.innerHTML = "";

    if (!state.goals.length) {
      container.innerHTML = `<div class="small-text">Još nema definisanih ciljeva. Prvi cilj obično bude Stabilizacija (izlazak iz minusa) ili mali sigurnosni fond.</div>`;
      return;
    }

    state.goals.forEach((g) => {
      const needed = Math.max(g.target - g.current, 0);
      const months = Math.max(g.months, 1);
      const monthlyTempo = needed / months;

      const div = document.createElement('div');
      div.className = "metric-box";
      div.style.background = "#ffffff";
      div.style.borderLeftColor = g.priority === "HITNO" ? "var(--danger)" : "var(--primary)";

      div.innerHTML = `
        <div class="flex-row">
          <strong>${g.name}</strong>
          <span class="badge ${priorityBadgeClass(g.priority)}">${g.priority}</span>
        </div>
        <div class="small-text" style="margin-top:3px;">
          Ciljni iznos: <strong>${fmt(g.target)} RSD</strong> · Već izdvojeno:
          <strong>${fmt(g.current)} RSD</strong>
        </div>
        <div class="small-text">
          Preostalo: <strong>${fmt(needed)} RSD</strong> · Rok: <strong>${months} meseci</strong>
        </div>
        <div style="margin-top:4px; font-size:13px; font-weight:600;">
          Potreban tempo: <span style="color:var(--warning);">${fmt(monthlyTempo)} RSD / mesec</span>
        </div>
        <div class="action-links" style="margin-top:4px;">
          <span class="action-link" data-edit="${g.id}">✏️ Ispravi</span>
          <span class="action-link danger" data-delete="${g.id}">🗑️ Obriši</span>
        </div>
      `;
      container.appendChild(div);
    });
  }

  // Goal edit/delete handlers
  function onGoalsClick(e) {
    const editId = e.target.getAttribute('data-edit');
    const delId = e.target.getAttribute('data-delete');
    if (editId) {
      editGoal(editId);
    } else if (delId) {
      deleteGoal(delId);
    }
  }

  function editGoal(id) {
    const g = state.goals.find(x => x.id === id);
    if (!g) return;
    const name = prompt("Novi naziv cilja:", g.name) || g.name;
    const target = Number(prompt("Novi ciljni iznos (RSD):", g.target)) || g.target;
    const current = Number(prompt("Već izdvojeno (RSD):", g.current)) || g.current;
    const months = Number(prompt("Rok u mesecima:", g.months)) || g.months;
    const priority = prompt("Prioritet (HITNO / VAŽNO / NORMALNO):", g.priority) || g.priority;

    g.name = name;
    g.target = target;
    g.current = current;
    g.months = months;
    g.priority = ["HITNO","VAŽNO","NORMALNO"].includes(priority) ? priority : g.priority;

    renderGoals();
  }

  function deleteGoal(id) {
    if (!confirm("Sigurno želiš da obrišeš ovaj cilj?")) return;
    state.goals = state.goals.filter(g => g.id !== id);
    renderGoals();
  }

  // Goal sustainability check
  function analyzeGoalSustainability() {
    const name = (document.getElementById('new-goal-name').value || "Novi cilj").trim();
    const amount = Number(document.getElementById('new-goal-amount').value) || 0;
    const months = Math.max(Number(document.getElementById('new-goal-months').value) || 1, 1);
    const priority = document.getElementById('new-goal-priority').value;
    const resultBox = document.getElementById('sustainability-result');
    resultBox.classList.remove('hidden');

    if (amount <= 0) {
      resultBox.className = "advisor-box warning";
      resultBox.innerHTML = `
        <h4>📌 Potreban je iznos cilja</h4>
        <p>Unesi realan iznos koji želiš da obezbediš da bismo mogli da testiramo plan.</p>
      `;
      return;
    }

    const neededMonthly = amount / months;
    const free = state.freeCashFlow;

    if (neededMonthly > free) {
      // Plan NIJE održiv
      resultBox.className = "advisor-box danger";
      const gap = neededMonthly - free;
      const recommendedMonths = free > 0 ? (amount / free) : months * 2;

      resultBox.innerHTML = `
        <h4>🔴 PLAN NIJE ODRŽIV</h4>
        <p>
          Da bi ostvarila cilj <strong>„${name}“</strong> u roku od
          <strong>${months} meseci</strong>, potrebno je izdvajanje od
          <strong>${fmt(neededMonthly)} RSD</strong> mesečno.<br>
          Tvoja trenutna čista fleksibilna dobit iznosi oko
          <strong>${fmt(free)} RSD</strong> mesečno. Finansijski jaz je
          približno <strong>${fmt(gap)} RSD</strong> svakog meseca.
        </p>
        <hr style="border:0;border-top:1px solid #fed7d7;margin:8px 0;">
        <strong>💡 REŠENJA I ALTERNATIVE (CFO savjetnik):</strong><br>
        • <strong>Opcija A – Povećanje prihoda:</strong>
          potrebno je generisati oko <strong>${fmt(gap)} RSD</strong> dodatnog prihoda mesečno
          (honorarni poslovi, gigovi).<br>
        • <strong>Opcija B – Pomeranje roka:</strong>
          ako pomerimo rok na oko <strong>${Math.ceil(recommendedMonths)} meseci</strong>,
          plan postaje realističan bez upadanja u minus.<br>
        • <strong>Opcija C – Smanjenje fleksibilne potrošnje:</strong>
          deo jaza može da se pokrije smanjenjem „skrivene potrošnje“
          (dostave, pretplate, sitne kupovine koje se ponavljaju).<br><br>
        <span class="small-text">
          ⚠️ <em>Napomena o kreditu:</em> kredit/pozajmica je ozbiljan alat i
          stvara novu mesečnu obavezu. FLOW ga ne preporučuje kao prvo rešenje
          za ovaj cilj. Tek ako ni povećanje prihoda ni pomeranje roka nisu
          realni, kredit može biti tema za posebnu analizu.</span>
      `;
    } else {
      resultBox.className = "advisor-box success";
      resultBox.innerHTML = `
        <h4>🟢 PLAN JE ODRŽIV</h4>
        <p>
          Potreban mesečni tempo od <strong>${fmt(neededMonthly)} RSD</strong> uklapa se
          u tvoju slobodnu dobit od oko <strong>${fmt(free)} RSD</strong> mesečno.<br>
          Ovaj cilj možeš da aktiviraš bez direktnog ugrožavanja bazične stabilnosti
          domaćinstva. I dalje će biti prostora za male šokove i normalan život.
        </p>
        <p class="small-text">
          Ako želiš, možeš ovaj cilj dodati u listu aktivnih ciljeva i pratiti ga kao
          „${priority}“ prioritet.
        </p>
      `;
    }
  }

  // Emergency Flow
  function executeEmergencyFlow() {
    const name = (document.getElementById('shock-name').value || "Neplanirani trošak").trim();
    const amount = Number(document.getElementById('shock-amount').value) || 0;
    const resBox = document.getElementById('emergency-result');
    resBox.classList.remove('hidden');

    if (amount <= 0) {
      resBox.className = "advisor-box warning";
      resBox.innerHTML = `
        <h4>📌 Potreban je iznos troška</h4>
        <p>Unesi realan iznos šoka da bismo mogli da simuliramo posledice.</p>
      `;
      return;
    }

    let report = `<h4>⚡ EMERGENCY FLOW — rezultat za: ${name}</h4>`;
    // 1. Tekući račun
    if (state.balance >= amount) {
      state.balance -= amount;
      resBox.className = "advisor-box success";
      report += `
        <p>
          🟢 <strong>Tekući račun je apsorbovao šok.</strong><br>
          Trošak od <strong>${fmt(amount)} RSD</strong> je pokriven direktno iz tekuće likvidnosti.
          Tvoji ciljevi i Zaštitni fond ostaju netaknuti.<br><br>
          Ovo je idealan scenario: sistem je podneo udar bez promene rokova ciljeva.
        </p>
      `;
    }
    // 2. Tekući + zaštitni
    else if (state.balance + state.shield >= amount) {
      const remaining = amount - state.balance;
      state.balance = 0;
      state.shield -= remaining;
      resBox.className = "advisor-box warning";
      report += `
        <p>
          🛡️ <strong>Aktivirana je zaštitna brana.</strong><br>
          Tekući račun je ispražnjen, a preostalih
          <strong>${fmt(remaining)} RSD</strong> je povučeno iz Zaštitnog fonda.<br><br>
          Rokovi tvojih ciljeva ostaju isti, ali nivo rezerve je sada niži.
          Sledećih par meseci pametno je da deo slobodne dobiti usmeriš u obnovu
          Zaštitnog fonda.
        </p>
      `;
    }
    // 3. Udar i na ciljeve
    else {
      resBox.className = "advisor-box danger";
      const totalLiquidity = state.balance + state.shield;
      let remainingShock = amount - totalLiquidity;
      state.balance = 0;
      state.shield = 0;

      // biramo cilj najnižeg prioriteta
      const priorities = ["NORMALNO", "VAŽNO", "HITNO"];
      let lowPriorityGoal = null;
      for (let p of priorities) {
        lowPriorityGoal = state.goals.find(g => g.priority === p);
        if (lowPriorityGoal) break;
      }

      report += `
        <p>
          🔴 <strong>Kritičan proboj odbrane.</strong><br>
          Sva likvidnost i zaštitni fond su potrošeni za pokriće dela šoka
          (<strong>${fmt(totalLiquidity)} RSD</strong>).<br>
          Preostali neabsorbowani deo šoka iznosi
          <strong>${fmt(remainingShock)} RSD</strong>.
        </p>
      `;

      if (lowPriorityGoal) {
        const oldMonths = lowPriorityGoal.months;
        const tempo = Math.max(state.freeCashFlow, 1);
        const addedMonths = Math.ceil(remainingShock / tempo);
        lowPriorityGoal.months += addedMonths;

        report += `
          <hr style="border:0;border-top:1px solid #fed7d7;margin:8px 0;">
          <p>
            ⚠️ <strong>Automatski rerouting ciljeva:</strong><br>
            FLOW je štitio ciljeve višeg prioriteta, pa je morao da promeni rutu
            cilja nižeg prioriteta:
          </p>
          <ul>
            <li>Cilj: <em>${lowPriorityGoal.name}</em></li>
            <li>Originalni rok: <strong>${oldMonths} meseci</strong></li>
            <li>Novi procenjeni rok: <strong>${lowPriorityGoal.months} meseci</strong></li>
          </ul>
          <p>
            Neplanirani trošak je praktično <strong>pomakao ovaj cilj za +${addedMonths} meseci</strong>.
            Ovo nije kazna, već realan prikaz posledice: deo slobodne dobiti
            će neko vreme ići na pokrivanje šoka.
          </p>
        `;
      } else {
        report += `
          <p>
            Nema ciljeva nižeg prioriteta koje možemo pomeriti. Plan je trenutno
            <strong>preopterećen</strong> i biće potrebno ozbiljnije povećanje prihoda
            (gigovi, dodatni posao) ili redefinisanje postojećih ciljeva.
          </p>
        `;
      }
    }

    renderDashboard();
    renderGoals();
    evaluateFinancialHealth();
    resBox.innerHTML = report;
  }

  // Gigs
  function injectGigEarnings(amount, name) {
    state.balance += amount;
    renderDashboard();
    evaluateFinancialHealth();
    alert(`🚀 Uspešno ubrizgavanje keša! Zaradila si ${fmt(amount)} RSD na lokaciji „${name}“.\n\nTekući račun je osvežen, a otpornost sistema je poboljšana.`);
  }

  // Tabs
  function switchTab(tabId, btn) {
    document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
    document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('active'));
    document.getElementById('tab-' + tabId).classList.remove('hidden');
    if (btn) btn.classList.add('active');
  }

  // Bindings
  function bindEvents() {
    document.getElementById('btn-init').addEventListener('click', initializeCFO);

    document.querySelectorAll('.tab-btn').forEach(btn => {
      btn.addEventListener('click', () => {
        const tab = btn.getAttribute('data-tab');
        switchTab(tab, btn);
      });
    });

    document.getElementById('btn-check-goal').addEventListener('click', analyzeGoalSustainability);
    document.getElementById('btn-emergency').addEventListener('click', executeEmergencyFlow);

    document.getElementById('goals-container').addEventListener('click', onGoalsClick);

    document.querySelectorAll('[data-gig]').forEach(btn => {
      btn.addEventListener('click', () => {
        const gig = btn.getAttribute('data-gig');
        if (gig === 'eko') {
          injectGigEarnings(11000, "Eko‑Farma Kosmaj");
        } else if (gig === 'logistika') {
          injectGigEarnings(4500, "Sezonska logistika");
        }
      });
    });
  }

  bindEvents();
})();
